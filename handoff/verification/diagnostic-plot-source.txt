  nlag = ifelse(S<7, 20, 3*S); nlag = min(nlag, 52)
  ppq  = p+q+P+Q - sum(!is.na(fixed)) + abs(fitdf)
  if (nlag < ppq + 8) nlag = ppq + 8
  for (i in (ppq+1):nlag) {
    u = Box.test(rs, i, type="Ljung-Box", fitdf=fitdf)$statistic
    pval[i] = pchisq(u, i-ppq, lower.tail=FALSE)
  }
